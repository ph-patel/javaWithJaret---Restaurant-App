package restaurant;

import java.net.URL;
import java.util.ResourceBundle;
import javafx.event.ActionEvent;
import javafx.fxml.FXML;
import javafx.fxml.Initializable;
import javafx.scene.control.Button;
import javafx.scene.control.CheckBox;
import javafx.scene.control.ComboBox;
import javafx.scene.control.DatePicker;
import javafx.scene.control.Label;
import javafx.scene.control.RadioButton;
import javafx.scene.control.TextField;
import javafx.scene.control.ToggleGroup;
import javafx.scene.image.Image;
import javafx.scene.image.ImageView;

/**
 *
 * @author Priyesh Hingrajia
 */
public class FXMLDocumentController implements Initializable { 
    
      @FXML
    private TextField resNameTextField;

    @FXML
    private TextField addressTextField;

    @FXML
    private ComboBox<String> cityComboBox;

    @FXML
    private TextField phoneTextField;

    @FXML
    private DatePicker startDate;

    @FXML
    private RadioButton burgerRadioButton;

    @FXML
    private RadioButton pizzaRadioButton;

    @FXML
    private RadioButton burritoRadioButton;

    @FXML
    private RadioButton subRadioButton;
    
    private ToggleGroup setResType ;

    @FXML
    private CheckBox uberEatsCheckBox;

    @FXML
    private CheckBox skipTheDishesCheckBox;

    @FXML
    private ImageView photo;

    @FXML
    private Button saveButton;

    @FXML
    private Label errMsg;

    @FXML
    void radioBtnChanged(ActionEvent event) {
          if (burritoRadioButton.isSelected())
              photo.setImage(new Image("/images/burrito.png")); 
          else if (burgerRadioButton.isSelected())
              photo.setImage(new Image("/images/burger.png")); 
          else if (subRadioButton.isSelected())
              photo.setImage(new Image("/images/sub.png"));
          else if (pizzaRadioButton.isSelected())
              photo.setImage(new Image("/images/pizza.png")); 
          else 
              photo.setImage(new Image("/images/default.jpg")); 
              
    }

    @FXML
    void saveButtonPressed(ActionEvent event) {
        if (this.startDate.getValue() != null)
        {
            try{
                
               
                Register newRestaurant = new Register(
                                        this.resNameTextField.getText(),
                                        this.addressTextField.getText(),
                                        this.cityComboBox.getValue(),
                                        this.burritoRadioButton.getText(),
                                        this.uberEatsCheckBox.getText(),
                                        this.phoneTextField.getText(),
                                        this.startDate.getValue());
                System.out.println(newRestaurant);
            } catch (IllegalArgumentException e)
            {
                errMsg.setText(e.getMessage());
            }
        }
        else
        {
            errMsg.setText("You need to set the start date");
        }
    }

    
    @Override
    public void initialize(URL url, ResourceBundle rb) {
         
        /**
         * this code makes the error message label disappear until error is triggered
         */
        errMsg.setText("");
        
        
        /**
         * this code configure the combo box and populate it with barrie , Toronto and Orillia .
         */
        cityComboBox.getItems().addAll("Barrie" , "Toronto" , "Orillia");
        cityComboBox.setValue("Barrie");
        
        /**
         * this code setup the radio button and introduce the toggle group
         */
        setResType = new ToggleGroup() ;
        this.burgerRadioButton.setToggleGroup(setResType);
        this.burritoRadioButton.setToggleGroup(setResType);
        this.pizzaRadioButton.setToggleGroup(setResType);
        this.subRadioButton.setToggleGroup(setResType);
        
    }     
    
    
    
}
